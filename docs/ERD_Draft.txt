    Users {
        INT UserID PK
        VARCHAR FirstName
        VARCHAR LastName
        VARCHAR Email
        VARCHAR PasswordHash
        VARCHAR UserRole
        VARCHAR Gender
        DATE DateOfBirth
        VARCHAR PhoneNumber
        VARCHAR EmergencyContactName
        VARCHAR EmergencyContactPhone
        TEXT MedicalConditions
        DATETIME RegistrationDate
    }

    Events {
        INT EventID PK
        INT OrganiserID FK
        VARCHAR EventName
        TEXT Description
        VARCHAR EventType
        DATE EventDate
        TIME StartTime
        VARCHAR LocationCity
        VARCHAR LocationProvince
        INT MaxCapacity
        DATE RegistrationDeadline
        BIT IsActive
        DATETIME CreatedAt
    }

    EventRoutes {
        INT RouteID PK
        INT EventID FK
        VARCHAR RouteName
        DECIMAL TotalDistanceKm
        INT ElevationGainMeters
        INT WaterPointsCount
        VARCHAR RouteMapUrl
    }

    Categories {
        INT CategoryID PK
        INT EventID FK
        VARCHAR CategoryName
        DECIMAL DistanceKm
        DECIMAL EntryFeeZAR
        INT MinAge
        INT MaxAge
        VARCHAR GenderRestriction
    }

    Enrolments {
        INT EnrolmentID PK
        INT ParticipantID FK
        INT CategoryID FK
        INT RaceNumber
        VARCHAR PaymentStatus
        DATETIME EnrolmentDate
    }

    Results {
        INT ResultID PK
        INT EnrolmentID FK
        TIME FinishTime
        INT OverallPosition
        INT CategoryPosition
        VARCHAR RaceStatus
        DATETIME CapturedAt
    }

    Users ||--o{ Events : "Organiser creates"
    Events ||--o{ EventRoutes : "has"
    Events ||--o{ Categories : "has"
    Users ||--o{ Enrolments : "Participant has"
    Categories ||--o{ Enrolments : "receives"
    Enrolments ||--|| Results : "generates"
